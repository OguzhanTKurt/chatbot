        async function sendDataTon8n() {
            const fileInput = document.getElementById('fileInput');
            const promptInput = document.getElementById('promptInput').value;
            const resultArea = document.getElementById('resultArea');
            const submitBtn = document.getElementById('submitBtn');
            const spinner = document.getElementById('spinner');
            const btnText = document.getElementById('btnText');

            if (fileInput.files.length === 0) {
                alert("Lütfen işleme başlamadan önce bir dosya seçin.");
                return;
            }
            if (promptInput.trim() === "") {
                alert("Lütfen yapay zeka için bir komut (prompt) girin.");
                return;
            }

            submitBtn.disabled = true;
            spinner.style.display = "block";
            btnText.innerText = "İşleniyor...";
            resultArea.innerHTML = "<em>Yapay zeka dosyayı inceliyor, lütfen bekleyin... (Bu işlem donanıma bağlı olarak biraz sürebilir)</em>";

            const formData = new FormData();
            const secilenDosya = fileInput.files[0];
            const dosyaUzantisi = secilenDosya.name.split('.').pop().toLowerCase();

            formData.append("file", secilenDosya); 
            formData.append("prompt", promptInput);
            formData.append("fileType", dosyaUzantisi); 

            // DİKKAT: URL'deki "webhook-test" kısmını üretim ortamında "webhook" yapmayı unutma!
            const n8nWebhookURL = "http://localhost:5678/webhook-test/ai-rag-analiz"; 

            try {
                const response = await fetch(n8nWebhookURL, {
                    method: "POST",
                    body: formData
                });

                if (response.ok) {
                    // YENİ EKLEME: n8n'den gelen cevabın formatını akıllıca kontrol et
                    const contentType = response.headers.get("content-type");
                    let resultData = "";

                    if (contentType && contentType.includes("application/json")) {
                        // Eğer n8n cevabı JSON objesi olarak gönderdiyse içindeki metni bulur
                        const jsonData = await response.json();
                        resultData = jsonData.text || jsonData.data || JSON.stringify(jsonData);
                    } else {
                        // Eğer n8n cevabı düz metin gönderdiyse doğrudan okur
                        resultData = await response.text(); 
                    }
                    
                    resultArea.innerText = resultData;
                } else {
                    resultArea.innerHTML = `<span style="color: red;">Sunucu Hatası: ${response.status} - İşlem reddedildi. Lütfen n8n Webhook düğümünde CORS'un açık olduğundan emin ol.</span>`;
                }
            } catch (error) {
                console.error("Fetch Hatası:", error);
                resultArea.innerHTML = `<span style="color: red;">Bağlantı hatası! n8n'e ulaşılamıyor.<br><br>Olası Sebepler:<br>1- URL'de "webhook-test" yazıyor ama n8n'de 'Test Workflow' butonuna basmadın.<br>2- Tarayıcı CORS politikası sebebiyle isteği engelledi.</span>`;
            } finally {
                submitBtn.disabled = false;
                spinner.style.display = "none";
                btnText.innerText = "Analiz Et";
            }
        }
    </script>
